$projects = @(@('DotGram.Sql', 'src\DotGram.Sql\DotGram.Sql.csproj'), @('DotGram.ExpressionLanguage', 'src\DotGram.ExpressionLanguage\DotGram.ExpressionLanguage.csproj'))
$rows = @()

function Measure-Build([string]$label) {
	foreach ($p in $projects) {
		$log = dotnet build (Join-Path $tree $p[1]) -c Release -v:quiet -m:1 -nodeReuse:false -p:UseSharedCompilation=false -t:Rebuild 2>&1
		if ($LASTEXITCODE -ne 0) { throw "$($p[0]) did not build ($label):`n$(($log | Select-Object -Last 12) -join "`n")" }

		foreach ($tfm in 'net10.0', 'netstandard2.0') {
			$dll = Join-Path (Split-Path (Join-Path $tree $p[1])) "bin\Release\$tfm\$($p[0]).dll"
			$bytes = (Get-Item $dll).Length
			$zip = [IO.MemoryStream]::new()
			$gz = [IO.Compression.GZipStream]::new($zip, [IO.Compression.CompressionLevel]::Optimal, $true)
			$raw = [IO.File]::ReadAllBytes($dll)
			$gz.Write($raw, 0, $raw.Length)
			$gz.Dispose()
			$script:rows += [pscustomobject]@{ Label = $label; Assembly = $p[0]; Framework = $tfm; Bytes = $bytes; Gzip = $zip.Length }
		}
	}
}

Measure-Build 'as it is'
