using System.Text;

using Mono.Cecil;

// attrcount <dll>...: every attribute whose name is GramSource or Gram (or GramInclude, GramOptions), on every type, with the length of each string argument (characters and UTF-8 bytes).
if (args.Length > 0 && args[0] == "--deflate")
{
	// The grammar texts on the classes, each deflated on its own (what the embedded PDB does to a generated source that carries it), and the sum.
	long sum = 0, raw = 0;

	foreach (var type in ModuleDefinition.ReadModule(args[1]).GetTypes())
		foreach (var attribute in type.CustomAttributes.Where(static one => one.AttributeType.Name == "GramSourceAttribute"))
		{
			var text = Encoding.UTF8.GetBytes((string)attribute.ConstructorArguments[0].Value);
			using var packed = new MemoryStream();

			using (var deflate = new System.IO.Compression.DeflateStream(packed, System.IO.Compression.CompressionLevel.Optimal, true))
				deflate.Write(text, 0, text.Length);

			Console.WriteLine($"  {type.Name,-24} {text.Length,10:N0} bytes, deflated {packed.Length,9:N0} ({100.0 * packed.Length / text.Length:F1}%)");
			sum += packed.Length;
			raw += text.Length;
		}

	Console.WriteLine($"  together {raw:N0} bytes, deflated {sum:N0} ({100.0 * sum / raw:F1}%)");

	return;
}

if (args.Length > 0 && args[0] == "--pdb")
{
	foreach (var one in args.Skip(1))
		Pdb.Run(one);

	return;
}

foreach (var path in args)
{
	var module = ModuleDefinition.ReadModule(path);
	long total = 0;

	Console.WriteLine($"== {Path.GetFileName(path)} ({new FileInfo(path).Length:N0} bytes on disk)");

	foreach (var type in module.GetTypes())
	{
		foreach (var attribute in type.CustomAttributes)
		{
			var name = attribute.AttributeType.Name;

			if (name is not ("GramSourceAttribute" or "GramAttribute"))
				continue;

			foreach (var argument in attribute.ConstructorArguments.Where(static one => one.Value is string))
			{
				var text  = (string)argument.Value;
				var bytes = Encoding.UTF8.GetByteCount(text);

				if (bytes > 200 || name == "GramSourceAttribute")
				{
					Console.WriteLine($"  {name,-20} on {type.FullName,-60} {text.Length,9:N0} chars {bytes,9:N0} bytes UTF-8   {text[..Math.Min(48, text.Length)].Replace('\n', ' ').Replace('\r', ' ')}...");

					if (name == "GramSourceAttribute")
						total += bytes;
				}
			}
		}
	}

	Console.WriteLine($"  GramSource bytes in total: {total:N0}");
}
