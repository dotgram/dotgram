=== s01-feed3
Feed    : @string[] = Header & Row* recover eos => @("bad " + parserLine + ":" + parserFailureColumn) & Trailer & eof
Header  : @string   = "H|" & t: Text & eos => @("head " + t)
Row     : @string   = n: Number & '|' & t: Text & eos => @(n + "=" + t)
Trailer : @string   = "T|" & n: Number & eos => @("tail " + n)
Number  = ['0'..'9']+
Text    = [^ '|']+

=== s02-feed-alts
Feed    : @string[] = Header & Row* recover eos => @("bad " + parserLine + ":" + parserFailureColumn) & Trailer & eof
Header  : @string   = "H|" & t: Text & eos => @("head " + t)
Row     : @string   = a: Trade => @(a) | b: Quote => @(b) | c: Note => @(c)
Trade   : @string   = "TR|" & n: Number & '|' & p: Price & o: Opt & eos => @("trade " + n + " " + p + " [" + o + "]")
Quote   : @string   = "QU|" & s: Sym & '|' & q: Quoted & eos => @("quote " + s + " " + q)
Note    : @string   = "NO|" & t: Rest & eos => @("note " + t)
Opt     = ('|' & Text)?
Price   = ['0'..'9']+ & ('.' & ['0'..'9']{1,4})?
Sym     = ['A'..'Z']{1,5}
Quoted  = '"' & ([^ '"'] | "\"\"")* & '"'
Rest    = any*
Trailer : @string   = "T|" & n: Number & eos => @("tail " + n)
Number  = ['0'..'9']+
Text    = [^ '|']+
